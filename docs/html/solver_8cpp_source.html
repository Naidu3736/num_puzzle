125 naidu::Board& node,
126 std::uint16_t g,
127 std::uint32_t threshold,
128 Context& ctx
129 ) {
130 ++ctx.nodes_expanded;
131
132 if (ctx.config.cancel_requested && ctx.config.cancel_requested())
133 return std::nullopt;
134
135 if (ctx.config.max_nodes != 0 && ctx.nodes_expanded > ctx.config.max_nodes)
136 return std::nullopt;
137
138 auto f = ctx.config.eval_fn(node, ctx.goal_pos, g);
139 if (f > threshold)
140 return f;
141
142 if (node == ctx.goal) {
143 ctx.found = true;
144 ctx.solution_cost = g;
145 return std::nullopt;
146 }
147
148 std::optional<std::uint32_t> min;
149
150 for (const auto d : node.legal_moves()) {
151 if (!ctx.path.empty() && d == naidu::opposite(ctx.path.back()))
152 continue;
153
154 std::uint16_t cost = 0;
155 if (!try_entry(node, d, ctx, cost))
156 continue;
157
158 const std::uint16_t new_g = static_cast<std::uint16_t>(g + cost);
159 const auto t = search(node, new_g, threshold, ctx);
160
161 if (ctx.found)
162 return std::nullopt;
163
164 leave(node, d, ctx);
165
166 if (t && (!min || *t < *min))
167 min = t;
168 }
169
170 return min;
