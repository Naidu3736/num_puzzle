191DifficultyMetrics AnalysisTab::run_difficulty_batch(int moves_count, int instances, std::uint8_t dim, int max_attempts, bool rand_avoid, const naidu::solver::Config& config) {
192 DifficultyMetrics metrics;
193 naidu::randomizer::Config rand_config;
194 rand_config.moves = static_cast<std::uint32_t>(moves_count);
195 rand_config.max_attempts = static_cast<std::uint32_t>(max_attempts == 0 ? 50000 : max_attempts);
196 rand_config.avoid_revisits = rand_avoid;
197
198 QList<int> tasks;
199 for (int i = 0; i < instances; ++i) tasks.append(i);
200
201 QMutex mutex;
202 int instance_counter = 0;
203
204 QtConcurrent::blockingMap(tasks, [&](int) {
205 std::random_device rd;
206 std::mt19937 rng(rd());
207
208 naidu::Board initial = naidu::randomizer::generate(dim, rand_config, rng);
209 naidu::Board goal(dim);
210
211 if (initial.is_solvable()) {
212 auto start = std::chrono::high_resolution_clock::now();
213 auto solution = naidu::solver::ida(initial, goal, config);
214 auto end = std::chrono::high_resolution_clock::now();
215 auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(end - start).count();
216
217 QMutexLocker locker(&mutex);
218 int current_id = instance_counter++;
219 if (solution.found) {
220 metrics.successes++;
221 metrics.execution_times.append(QPointF(current_id, static_cast<qreal>(duration)));
222 metrics.total_cost += solution.cost;
223 metrics.total_mixing += moves_count;
224 } else {
225 metrics.failures++;
226 }
227 } else {
228 QMutexLocker locker(&mutex);
229 metrics.failures++;
230 }
231 });
232
233 return metrics;
234}
