13 inline void check_default(const naidu::Board& b, std::size_t dim) {
14 CHECK(b.dim() == dim);
15
16 for (std::size_t i = 0; i < b.tile_count(); ++i) {
17 if (i == b.tile_count() - 1)
18 CHECK(b.at(i) == 0);
19 else
20 CHECK(b.at(i) == i + 1);
21 }
22
23 CHECK(b.empty_index() == b.tile_count() - 1);
24 CHECK(b.is_solved());
25 }
