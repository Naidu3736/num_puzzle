106std::optional<naidu::Board> PuzzleTab::parse_board_input(const QString& text, std::uint8_t dim) const {
107 QStringList parts = text.split(',', Qt::SkipEmptyParts);
108 std::vector<std::uint8_t> tiles;
109 tiles.reserve(parts.size());
110 for (const QString& part : parts) {
111 bool ok = false;
112 int val = part.trimmed().toInt(&ok);
113 if (!ok || val < 0 || val >= dim * dim) return std::nullopt;
114 tiles.push_back(static_cast<std::uint8_t>(val));
115 }
116 try {
117 return naidu::Board(tiles);
118 } catch (...) {
119 return std::nullopt;
120 }
121}

123void PuzzleTab::on_solve_clicked() {
124 if (m_solver_watcher.isRunning()) {
125 QMessageBox::warning(this, "Aviso", "Ya hay una búsqueda en curso.");
126 return;
127 }
128
129 if (m_anim_timer.isActive()) {
130 m_anim_timer.stop();
131 }
132
133 QString initial_text = m_ui->initial_input->text().trimmed();
134 QString goal_text = m_ui->goal_input->text().trimmed();
135
136 if (initial_text.isEmpty()) {
137 QMessageBox::warning(this, "Error", "El campo de estado inicial no puede estar vacío.");
138 return;
139 }
140
141 QStringList initial_parts = initial_text.split(',', Qt::SkipEmptyParts);
142 int initial_total = initial_parts.size();
143 int initial_dim_int = static_cast<int>(std::round(std::sqrt(initial_total)));
144 if (initial_dim_int * initial_dim_int != initial_total || initial_dim_int < 3 || initial_dim_int > 15) {
145 QMessageBox::warning(this, "Error", "El tamaño del estado inicial no forma un tablero cuadrado válido.");
146 return;
147 }
148 std::uint8_t initial_dim = static_cast<std::uint8_t>(initial_dim_int);
149
150 auto initial_parsed = parse_board_input(initial_text, initial_dim);
151 if (!initial_parsed) {
152 QMessageBox::warning(this, "Error", "El estado inicial ingresado no es válido.");
153 return;
154 }
155
156 naidu::Board initial_board = *initial_parsed;
157 naidu::Board goal_board;
158
159 // Si el campo meta está vacío, se asume automáticamente la meta estándar por defecto
160 if (goal_text.isEmpty()) {
161 goal_board = naidu::Board(initial_dim);
162 } else {
163 QStringList goal_parts = goal_text.split(',', Qt::SkipEmptyParts);
164 int goal_total = goal_parts.size();
165 int goal_dim_int = static_cast<int>(std::round(std::sqrt(goal_total)));
166 if (goal_dim_int * goal_dim_int != goal_total || goal_dim_int < 3 || goal_dim_int > 15) {
167 QMessageBox::warning(this, "Error", "El tamaño del estado meta no forma un tablero cuadrado válido.");
168 return;
169 }
170 std::uint8_t goal_dim = static_cast<std::uint8_t>(goal_dim_int);
171
172 auto goal_parsed = parse_board_input(goal_text, goal_dim);
173 if (!goal_parsed) {
174 QMessageBox::warning(this, "Error", "El estado meta ingresado no es válido.");
175 return;
176 }
177 goal_board = *goal_parsed;
178 }
179
180 if (initial_board.dim() != goal_board.dim()) {
181 QMessageBox::warning(this, "Error", "El tablero inicial y el tablero meta deben tener exactamente la misma dimensión.");
182 return;
183 }
184
185 m_initial_board = initial_board;
186 m_goal_board = goal_board;
187
188 naidu::solver::Config config;
189 config.eval_fn = get_selected_evaluation();
190 config.max_nodes = static_cast<std::uint64_t>(m_ui->max_nodes_spin->value());
191 config.avoid_revisits = m_ui->avoid_revisits_solver_check->isChecked();
192 config.weighted_cost = m_ui->weighted_cost_check->isChecked();
193 config.cancel_requested = [this]() {
194 return m_solver_watcher.isCanceled();
195 };
196
197 m_ui->status_label->setText("Resolviendo en segundo plano...");
198 m_ui->solve_btn->setEnabled(false);
199 m_ui->stop_btn->setEnabled(true);
200
201 m_start_time = std::chrono::high_resolution_clock::now();
202
203 QFuture<naidu::solver::Solution> future = QtConcurrent::run(
204 [initial_board, goal_board, config]() {
205 return naidu::solver::ida(initial_board, goal_board, config);
206 }
207 );
208
209 m_solver_watcher.setFuture(future);
210}

212void PuzzleTab::on_solver_finished() {
213 m_ui->solve_btn->setEnabled(true);
214 m_ui->stop_btn->setEnabled(false);
215
216 if (m_solver_watcher.isCanceled()) {
217 return;
218 }
219
220 naidu::solver::Solution solution = m_solver_watcher.result();
221
222 auto end_time = std::chrono::high_resolution_clock::now();
223 auto duration_ms = std::chrono::duration_cast<std::chrono::milliseconds>(end_time - m_start_time).count();
224 m_ui->runtime_value_label->setText(QString::number(duration_ms) + " ms");
225
226 if (solution.found) {
227 m_ui->status_label->setText("¡Solución encontrada! Animando...");
228 m_ui->nodes_value_label->setText(QString::number(solution.nodes_expanded));
229 m_ui->cost_value_label->setText(QString::number(solution.cost));
230
231 QStringList path_list;
232 for (auto d : solution.moves) {
233 switch(d) {
234 case naidu::Direction::UP: path_list.append("U"); break;
235 case naidu::Direction::DOWN: path_list.append("D"); break;
236 case naidu::Direction::LEFT: path_list.append("L"); break;
237 case naidu::Direction::RIGHT: path_list.append("R"); break;
238 }
239 }
240
241 m_ui->path_content_label->setText(path_list.join(","));
242 m_ui->path_content_label->adjustSize();
243 m_ui->path_scroll_area->updateGeometry();
244
245 m_anim_moves = solution.moves;
246 on_replay_animation();
247 } else {
248 m_ui->status_label->setText("No se encontró solución (límite de nodos alcanzado).");
249 m_ui->nodes_value_label->setText(QString::number(solution.nodes_expanded));
250 m_ui->path_content_label->clear();
251 }
